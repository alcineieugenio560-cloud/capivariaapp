name: Build APK
on:
  push:
    branches: [main]
  workflow_dispatch:
permissions:
  contents: write
jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - id: p
        run: |
          D=$(dirname "$(find . -name settings.gradle.kts -not -path './.git/*' | head -1)")
          echo "Projeto encontrado em: $D"
          echo "dir=$D" >> "$GITHUB_OUTPUT"
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - name: Gerar icone com a logo da Capivaria
        run: |
          pip install pillow
          python3 "${{ steps.p.outputs.dir }}/tools/make_icons.py" "${{ steps.p.outputs.dir }}/app/src/main/res"
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17
      - uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: "8.7"
      - run: gradle assembleDebug --no-daemon -p "${{ steps.p.outputs.dir }}"
      - uses: actions/upload-artifact@v4
        with:
          name: capivaria-apk
          path: ${{ steps.p.outputs.dir }}/app/build/outputs/apk/debug/app-debug.apk
      - uses: softprops/action-gh-release@v2
        with:
          tag_name: v${{ github.run_number }}
          files: ${{ steps.p.outputs.dir }}/app/build/outputs/apk/debug/app-debug.apk
